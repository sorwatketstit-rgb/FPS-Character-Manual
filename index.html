Mouse Look Controller

▸

📜 ดูโค้ด C# : SimpleFPSCamera.cs
C# – SimpleFPSCamera.cs 
using UnityEngine;

public class SimpleFPSCamera : MonoBehaviour
{
    [Header("References")]
    [Tooltip("The player body transform that rotates left/right (yaw).")]
    public Transform player;

    [Tooltip("The transform that pitches up/down (looks up/down). Defaults to this object.")]
    public Transform cameraHolder;

    [Header("Mouse Settings")]
    [Range(1f, 1000f)]
    public float mouseSensitivity = 400f;

    [Tooltip("Invert Y-axis (up/down)")]
    public bool invertY = false;

    [Header("Camera Limits")]
    [Range(60f, 90f)]
    public float maxLookAngle = 90f;

    private float xRotation = 0f;
    private float yRotation = 0f;

    void Start()
    {
        Cursor.lockState = CursorLockMode.Locked;
        Cursor.visible = false;

        if (cameraHolder == null)
        {
            cameraHolder = transform;
        }

        if (player != null)
        {
            yRotation = player.eulerAngles.y;
        }
    }

    void Update()
    {
        if (Cursor.lockState == CursorLockMode.Locked)
        {
            HandleMouseLook();
        }
    }

    void HandleMouseLook()
    {
        float mouseX = Input.GetAxisRaw("Mouse X") * mouseSensitivity * Time.deltaTime;
        float mouseY = Input.GetAxisRaw("Mouse Y") * mouseSensitivity * Time.deltaTime;

        if (invertY) mouseY = -mouseY;

        xRotation -= mouseY;
        xRotation = Mathf.Clamp(xRotation, -maxLookAngle, maxLookAngle);

        yRotation += mouseX;

        cameraHolder.localRotation = Quaternion.Euler(xRotation, 0f, 0f);

        if (player != null)
        {
            player.rotation = Quaternion.Euler(0f, yRotation, 0f);
        }
    }

    public void SetSensitivity(float sensitivity)
    {
        mouseSensitivity = Mathf.Clamp(sensitivity, 1f, 1000f);
    }
}

Physics Movement Engine

ระบบเดิน วิ่ง กระโดด ด้วยแรงฟิสิกส์ AddForce(..., ForceMode.VelocityChange) ใน FixedUpdate

▸
Ground Check: ตรวจพื้นด้วย Raycast ระยะ groundDistance ลงไปยัง groundMask
▸
Extra Gravity Fall: เพิ่มแรงเร่งโน้มถ่วง gravityMultiplier ตอนร่วง ให้จังหวะตกลงพื้นแน่น ไม่ลอย
▸
Dynamic Drag: สลับแรงต้านอัตโนมัติ บนพื้นใช้ groundDrag กลางอากาศใช้ airDrag และ airControl
📜 ดูโค้ด C# : SimpleMovement.cs
C# – SimpleMovement.cs 
using UnityEngine;

[RequireComponent(typeof(Rigidbody))]
public class SimpleMovement : MonoBehaviour
{
    [Header("Movement Settings")]
    [Range(20f, 100f)] public float moveSpeed = 50f;
    [Range(1f, 500f)] public float maxSpeed = 150f;
    [Range(20f, 150f)] public float acceleration = 50f;
    [Range(0f, 1f)] public float airControl = 0.5f;

    [Header("Gravity Settings")]
    [Range(1f, 25f)] public float gravityMultiplier = 2.5f;

    [Header("Jump Settings")]
    [Range(1f, 20f)] public float jumpForce = 12f;

    [Header("Drag Settings")]
    [Range(0f, 20f)] public float groundDrag = 5f;
    [Range(0f, 20f)] public float airDrag = 0.5f;

    [Header("Ground Check")]
    public LayerMask groundMask;
    [Range(0.1f, 1f)] public float groundDistance = 0.2f;

    private Rigidbody rb;
    private CapsuleCollider capsule;

    private Vector3 moveInput;
    private bool isGrounded;

    void Start()
    {
        rb = GetComponent<Rigidbody>();
        capsule = GetComponentInChildren<CapsuleCollider>();

        rb.freezeRotation = true;
        rb.interpolation = RigidbodyInterpolation.Interpolate;
    }

    void Update()
    {
        CheckGround();
        HandleInput();
    }

    void FixedUpdate()
    {
        MovePlayer();
        ApplyExtraGravity();
        ApplyDrag();
        ClampSpeed();
    }

    void HandleInput()
    {
        float h = Input.GetAxisRaw("Horizontal");
        float v = Input.GetAxisRaw("Vertical");
        moveInput = (transform.right * h + transform.forward * v).normalized;

        if (Input.GetKeyDown(KeyCode.Space) && isGrounded)
        {
            Jump();
        }
    }

    void MovePlayer()
    {
        Vector3 target = moveInput * moveSpeed;
        Vector3 current = HorizontalVelocity();

        float control = isGrounded ? 1f : airControl;
        Vector3 change = (target - current) * control;
        change = Vector3.ClampMagnitude(change, acceleration * Time.fixedDeltaTime);

        rb.AddForce(change, ForceMode.VelocityChange);
    }

    void Jump()
    {
        rb.linearVelocity = new Vector3(rb.linearVelocity.x, 0f, rb.linearVelocity.z);
        rb.AddForce(Vector3.up * jumpForce, ForceMode.Impulse);
    }

    void CheckGround()
    {
        float dist = (capsule.height * 0.5f) + groundDistance;
        isGrounded = Physics.Raycast(transform.position, Vector3.down, dist, groundMask);
    }

    void ApplyExtraGravity()
    {
        if (!isGrounded && rb.linearVelocity.y < 0)
        {
            rb.AddForce(
                Vector3.down * (gravityMultiplier - 1f) * Physics.gravity.magnitude,
                ForceMode.Acceleration
            );
        }
    }

    void ApplyDrag()
    {
        rb.linearDamping = isGrounded ? groundDrag : airDrag;
    }

    void ClampSpeed()
    {
        Vector3 horiz = HorizontalVelocity();
        if (horiz.magnitude > maxSpeed)
        {
            Vector3 clamped = horiz.normalized * maxSpeed;
            rb.linearVelocity = new Vector3(clamped.x, rb.linearVelocity.y, clamped.z);
        }
    }

    Vector3 HorizontalVelocity()
    {
        return new Vector3(rb.linearVelocity.x, 0f, rb.linearVelocity.z);
    }
}

▸

using UnityEngine;

public class SimpleShooting : MonoBehaviour
{
    [Header("Shooting Settings")]
    [Tooltip("Camera to shoot from. Auto-assigned to Camera.main if left empty.")]
    public Camera playerCamera;

    [Tooltip("Maximum distance the raycast can travel")]
    [Range(0f, 1000f)]
    public float shootRange = 100f;

    [Tooltip("Layer mask for what can be hit")]
    public LayerMask hitLayers;

    void Start()
    {
        if (playerCamera == null)
        {
            playerCamera = Camera.main;
            if (playerCamera == null)
            {
                Debug.LogError("No camera found! Assign a camera to SimpleShooting.");
                enabled = false;
            }
        }
    }

    void Update()
    {
        if (Input.GetButtonDown("Fire1"))
        {
            Shoot();
        }
    }

    void Shoot()
    {
        Ray ray = playerCamera.ViewportPointToRay(new Vector3(0.5f, 0.5f, 0f));
        RaycastHit hit;

        if (Physics.Raycast(ray, out hit, shootRange, hitLayers))
        {
            Debug.Log($"{hit.collider.gameObject.name} got hit");
        }
    }
}

C# – ExtraMovement.cs 
using UnityEngine;

[RequireComponent(typeof(Rigidbody))]
[DefaultExecutionOrder(100)]
public class ExtraMovement : MonoBehaviour
{
    [Header("Coexistence")]
    [Tooltip("Your normal movement script. Disabled while sliding/wall running so it doesn't fight this script. Optional.")]
    public MonoBehaviour movementScriptToSuspend;

    [Header("Ground Check")]
    [Tooltip("Should match the ground layer your main movement script uses.")]
    public LayerMask groundMask;
    [Range(0.1f, 1f)] public float groundDistance = 0.2f;

    [Header("Double Jump")]
    public bool enableDoubleJump = true;
    [Tooltip("How many extra jumps are allowed while airborne (1 = a single double jump).")]
    [Range(0, 5)] public int maxAirJumps = 1;
    [Range(1f, 20f)] public float airJumpForce = 12f;

    [Header("Slide Settings")]
    public bool enableSlide = true;
    [Range(1f, 20f)] public float slideMinSpeed = 5f;
    [Range(0f, 75f)] public float slideBoost = 30f;
    [Range(0f, 5f)] public float slideDrag = 2f;
    [Range(0.4f, 1f)] public float slideHeightMultiplier = 0.6f;

    [Header("Slide Tilt")]
    [Tooltip("Reference top speed used to scale tilt strength. Set this to match your movement script's max speed.")]
    [Range(1f, 500f)] public float referenceMaxSpeed = 150f;
    [Range(0f, 60f)] public float slideTiltAngle = 25f;
    [Range(1f, 20f)] public float slideTiltSpeed = 12f;
    [Range(0f, 1f)] public float slideTiltStrength = 1f;

    [Header("Wall Run Settings")]
    public bool enableWallRun = true;
    [Tooltip("Layer(s) that count as a wall. Leave empty to reuse Ground Mask above.")]
    public LayerMask wallMask;
    [Tooltip("Caps how fast you fall while against a wall.")]
    [Range(1f, 100f)] public float wallRunFallClamp = 2f;
    [Range(1f, 50f)] public float wallJumpForce = 15f;
    [Range(0f, 0.5f)] public float wallJumpDuration = 0.2f;

    [Header("Wall Run Tilt")]
    public bool enableWallRunTilt = true;
    [Range(0f, 45f)] public float wallRunTiltAngle = 15f;

    [Header("Visuals (optional)")]
    [Tooltip("Visual model to tilt during slides/wall runs. Leave empty to skip tilting.")]
    public Transform playerModel;

    private Rigidbody rb;
    private CapsuleCollider capsule;

    private bool isGrounded;
    private bool isSliding;
    private bool isWallRunning;
    private int airJumpsUsed;
    private float movementLockTimer;

    private float normalHeight;
    private float slideHeight;
    private Vector3 normalCenter;
    private Vector3 slideCenter;
    private Vector3 slideDirection;
    private Vector3 wallNormal;

    void Start()
    {
        rb = GetComponent<Rigidbody>();
        capsule = GetComponentInChildren<CapsuleCollider>();

        if (capsule != null)
        {
            normalHeight = capsule.height;
            slideHeight = normalHeight * slideHeightMultiplier;
            normalCenter = capsule.center;
            slideCenter = new Vector3(normalCenter.x, slideHeight * 0.5f, normalCenter.z);
        }
    }

    void Update()
    {
        CheckGround();

        if (isGrounded)
        {
            airJumpsUsed = 0;
        }

        HandleWallJumpInput();
        HandleDoubleJumpInput();
        HandleSlideInput();

        if (isSliding)
        {
            HandleSlideTilt();
        }
        else if (isWallRunning && enableWallRunTilt)
        {
            HandleWallRunTilt();
        }
        else
        {
            ResetModelTilt();
        }

        UpdateSuspendedState();
    }

    void FixedUpdate()
    {
        HandleWallRunPhysics();

        if (isSliding)
        {
            rb.linearDamping = slideDrag;
        }

        if (movementLockTimer > 0f)
        {
            movementLockTimer -= Time.fixedDeltaTime;
        }
    }

    void UpdateSuspendedState()
    {
        if (movementScriptToSuspend == null) return;

        bool shouldSuspend = isSliding || isWallRunning || movementLockTimer > 0f;
        movementScriptToSuspend.enabled = !shouldSuspend;
    }

    void HandleDoubleJumpInput()
    {
        if (!enableDoubleJump) return;
        if (isGrounded || isWallRunning) return;

        if (Input.GetKeyDown(KeyCode.Space) && airJumpsUsed < maxAirJumps)
        {
            rb.linearVelocity = new Vector3(rb.linearVelocity.x, 0f, rb.linearVelocity.z);
            rb.AddForce(Vector3.up * airJumpForce, ForceMode.Impulse);
            airJumpsUsed++;
        }
    }

    void HandleSlideInput()
    {
        if (!enableSlide || capsule == null) return;

        if (Input.GetKeyDown(KeyCode.LeftShift))
        {
            Vector3 horiz = HorizontalVelocity();
            if (isGrounded && horiz.magnitude >= slideMinSpeed)
            {
                StartSlide(horiz);
            }
        }

        if (Input.GetKeyUp(KeyCode.LeftShift))
        {
            StopSlide();
        }
    }

    void StartSlide(Vector3 horizontalVelocity)
    {
        if (isSliding) return;

        isSliding = true;

        slideDirection = horizontalVelocity.normalized;
        rb.AddForce(slideDirection * slideBoost, ForceMode.VelocityChange);

        capsule.height = slideHeight;
        capsule.center = slideCenter;
    }

    void StopSlide()
    {
        if (!isSliding) return;
        if (!CanStandUp()) return;

        isSliding = false;

        capsule.height = normalHeight;
        capsule.center = normalCenter;
    }

    bool CanStandUp()
    {
        float castDist = normalHeight - slideHeight + 0.1f;
        return !Physics.SphereCast(
            transform.position,
            capsule.radius * 0.95f,
            Vector3.up,
            out _,
            castDist,
            groundMask
        );
    }

    void HandleSlideTilt()
    {
        if (!playerModel) return;

        Vector3 horiz = HorizontalVelocity();
        if (horiz.sqrMagnitude < 0.1f) return;

        float speed01 = Mathf.Clamp01(horiz.magnitude / referenceMaxSpeed);
        float tilt = slideTiltAngle * speed01 * slideTiltStrength;

        Vector3 axis = Vector3.Cross(horiz.normalized, Vector3.up);
        Quaternion target = Quaternion.AngleAxis(tilt, axis) * transform.rotation;

        playerModel.rotation = Quaternion.Slerp(playerModel.rotation, target, Time.deltaTime * slideTiltSpeed);
    }

    LayerMask EffectiveWallMask()
    {
        return wallMask.value != 0 ? wallMask : groundMask;
    }

    void HandleWallJumpInput()
    {
        if (!enableWallRun) return;
        if (!isWallRunning) return;

        if (Input.GetKeyDown(KeyCode.Space))
        {
            WallJump();
        }
    }

    void WallJump()
    {
        Vector3 dir = (wallNormal + Vector3.up).normalized;
        rb.linearVelocity = new Vector3(rb.linearVelocity.x, 0f, rb.linearVelocity.z);
        rb.AddForce(dir * wallJumpForce, ForceMode.Impulse);

        isWallRunning = false;
        movementLockTimer = wallJumpDuration;
    }

    void HandleWallRunPhysics()
    {
        if (!isWallRunning) return;

        if (rb.linearVelocity.y < 0)
        {
            rb.linearVelocity = new Vector3(
                rb.linearVelocity.x,
                Mathf.Max(rb.linearVelocity.y, -wallRunFallClamp),
                rb.linearVelocity.z
            );
        }
    }

    void HandleWallRunTilt()
    {
        if (!playerModel) return;

        Vector3 axis = Vector3.Cross(-wallNormal, Vector3.up);
        playerModel.rotation = Quaternion.Slerp(
            playerModel.rotation,
            Quaternion.AngleAxis(wallRunTiltAngle, axis) * transform.rotation,
            Time.deltaTime * 8f
        );
    }

    void ResetModelTilt(float speed = 10f)
    {
        if (!playerModel) return;
        playerModel.rotation = Quaternion.Slerp(playerModel.rotation, transform.rotation, Time.deltaTime * speed);
    }

    void CheckGround()
    {
        if (capsule == null) return;
        float dist = (capsule.height * 0.5f) + groundDistance;
        isGrounded = Physics.Raycast(transform.position, Vector3.down, dist, groundMask);
    }

    Vector3 HorizontalVelocity()
    {
        return new Vector3(rb.linearVelocity.x, 0f, rb.linearVelocity.z);
    }

    void OnCollisionStay(Collision c)
    {
        if (!enableWallRun) return;
        if (isGrounded) return;

        LayerMask mask = EffectiveWallMask();
        if (((1 << c.gameObject.layer) & mask) == 0) return;

        foreach (var contact in c.contacts)
        {
            float angle = Vector3.Angle(contact.normal, Vector3.up);
            if (angle > 70f && angle < 110f)
            {
                isWallRunning = true;
                wallNormal = contact.normal;
                return;
            }
        }

        isWallRunning = false;
    }

    void OnCollisionExit(Collision c)
    {
        LayerMask mask = EffectiveWallMask();
        if (((1 << c.gameObject.layer) & mask) != 0)
            isWallRunning = false;
    }
}

Juice & Weapon Feedback

ควบคุมแรงสะบัดของปืนทั้งเชิงมุมและตำแหน่ง พร้อมจัดการ Muzzle Flash, Decal Offset, Particle และเสียงปืนแบบไดนามิก

▸

▸
Muzzle Flash & Particle: เกิดที่จุด muzzlePoint และทำลายอัตโนมัติตามระยะเวลา Lifetime
▸
Decal Placement: วางรอยกระสุนตาม Normal พร้อมคำนวณ Offset ป้องกัน Z-Fighting และตั้ง Parent ติดตามวัตถุได้

using UnityEngine;

[DisallowMultipleComponent]
public class GunFeel : MonoBehaviour
{
    [Header("References")]
    [Tooltip("Camera to raycast from. Auto-assigned to Camera.main if left empty.")]
    [SerializeField] private Camera playerCamera;

    [Tooltip("Transform recoil is applied to — e.g. your gun model.")]
    [SerializeField] private Transform recoilPivot;

    [Tooltip("Empty transform at the gun barrel tip — muzzle flash and shot origin.")]
    [SerializeField] private Transform muzzlePoint;

    [Header("Fire Rate")]
    [Tooltip("If on, holding the fire button keeps firing at 'Fire Rate'. If off, each click fires once.")]
    public bool automaticFire = false;
    [Range(1f, 20f)] public float fireRate = 8f;

    [Header("Recoil Settings")]
    public bool enableRecoil = true;
    [Range(0f, 20f)] public float kickAngle = 3f;
    [Range(0f, 10f)] public float horizontalKick = 1f;
    [Range(0f, 0.5f)] public float kickBackDistance = 0.05f;
    [Range(1f, 50f)] public float snappiness = 12f;
    [Range(1f, 50f)] public float recoverySpeed = 8f;

    [Header("Audio Settings")]
    public bool enableAudio = true;
    [Tooltip("AudioSource to play gunshot sound. Auto-added if not assigned.")]
    [SerializeField] private AudioSource audioSource;
    public AudioClip gunshotClip;
    [Range(0f, 1f)] public float volume = 0.8f;
    [Tooltip("Random pitch variation range to prevent repetitive sound artifacts.")]
    [Range(0.8f, 1.2f)] public float minPitch = 0.95f;
    [Range(0.8f, 1.2f)] public float maxPitch = 1.05f;

    [Header("Muzzle Flash")]
    public bool enableMuzzleFlash = true;
    public GameObject muzzleFlashPrefab;
    [Range(0.05f, 1f)] public float muzzleFlashLifetime = 0.15f;

    [Header("Impact FX & Bullet Decals")]
    public bool enableBulletDecals = true;
    public GameObject decalPrefab;
    [Tooltip("Particle effect spawned at the bullet hit point (e.g. sparks, dust, blood).")]
    public GameObject impactParticlePrefab;
    [Range(0.05f, 5f)] public float impactParticleLifetime = 2f;
    [Range(0.001f, 0.05f)] public float decalOffset = 0.01f;
    public bool parentDecalToHitObject = true;
    [Range(0f, 120f)] public float decalLifetime = 30f;
    [Range(0f, 1000f)] public float shootRange = 100f;
    public LayerMask hitLayers = ~0;

    private Vector3 currentRotation;
    private Vector3 targetRotation;
    private Vector3 currentKickPos;
    private Vector3 targetKickPos;
    private Quaternion baseRotation = Quaternion.identity;
    private Vector3 basePosition = Vector3.zero;

    private float nextFireTime;

    private void Start()
    {
        InitializeReferences();
    }

    private void InitializeReferences()
    {
        if (playerCamera == null)
        {
            playerCamera = Camera.main;
        }

        if (recoilPivot != null)
        {
            baseRotation = recoilPivot.localRotation;
            basePosition = recoilPivot.localPosition;
        }
        else
        {
            Debug.LogWarning($"[GunFeel] 'recoilPivot' is not assigned on {gameObject.name}.", this);
        }

        if (enableAudio && audioSource == null)
        {
            if (!TryGetComponent(out audioSource))
            {
                audioSource = gameObject.AddComponent<AudioSource>();
            }
            audioSource.playOnAwake = false;
        }
    }

    private void Update()
    {
        HandleFiringInput();
        UpdateRecoil();
    }

    private void HandleFiringInput()
    {
        bool triggerPulled = automaticFire ? Input.GetButton("Fire1") : Input.GetButtonDown("Fire1");

        if (triggerPulled && Time.time >= nextFireTime)
        {
            Fire();
            nextFireTime = Time.time + (1f / Mathf.Max(0.01f, fireRate));
        }
    }

    private void Fire()
    {
        ApplyRecoilKick();

        if (enableAudio) PlayGunshotSound();
        if (enableMuzzleFlash) PlayMuzzleFlash();
        if (enableBulletDecals) CastRayAndSpawnImpactEffects();
    }

    private void ApplyRecoilKick()
    {
        if (!enableRecoil) return;

        targetRotation += new Vector3(-kickAngle, Random.Range(-horizontalKick, horizontalKick), 0f);
        targetKickPos += new Vector3(0f, 0f, -kickBackDistance);
    }

    private void UpdateRecoil()
    {
        if (recoilPivot == null) return;

        float dt = Time.deltaTime;

        currentRotation = Vector3.Lerp(currentRotation, targetRotation, snappiness * dt);
        targetRotation = Vector3.Lerp(targetRotation, Vector3.zero, recoverySpeed * dt);

        currentKickPos = Vector3.Lerp(currentKickPos, targetKickPos, snappiness * dt);
        targetKickPos = Vector3.Lerp(targetKickPos, Vector3.zero, recoverySpeed * dt);

        recoilPivot.localRotation = baseRotation * Quaternion.Euler(currentRotation);
        recoilPivot.localPosition = basePosition + currentKickPos;
    }

    private void PlayGunshotSound()
    {
        if (gunshotClip == null || audioSource == null) return;

        audioSource.pitch = Random.Range(minPitch, maxPitch);
        audioSource.PlayOneShot(gunshotClip, volume);
    }

    private void PlayMuzzleFlash()
    {
        if (muzzleFlashPrefab == null || muzzlePoint == null) return;

        GameObject flash = Instantiate(muzzleFlashPrefab, muzzlePoint.position, muzzlePoint.rotation, muzzlePoint);
        Destroy(flash, muzzleFlashLifetime);
    }

    private void CastRayAndSpawnImpactEffects()
    {
        if (playerCamera == null) return;

        Ray ray = playerCamera.ViewportPointToRay(new Vector3(0.5f, 0.5f, 0f));

        if (Physics.Raycast(ray, out RaycastHit hit, shootRange, hitLayers))
        {
            Vector3 spawnPosition = hit.point + (hit.normal * decalOffset);
            Quaternion rotation = Quaternion.LookRotation(hit.normal);
            Transform parent = parentDecalToHitObject ? hit.collider.transform : null;

            if (decalPrefab != null)
            {
                Quaternion decalRotation = rotation * Quaternion.Euler(0f, 0f, Random.Range(0f, 360f));
                GameObject decal = Instantiate(decalPrefab, spawnPosition, decalRotation, parent);

                if (decalLifetime > 0f)
                {
                    Destroy(decal, decalLifetime);
                }
            }

            if (impactParticlePrefab != null)
            {
                GameObject particle = Instantiate(impactParticlePrefab, spawnPosition, rotation, parent);

                if (impactParticleLifetime > 0f)
                {
                    Destroy(particle, impactParticleLifetime);
                }
            }
        }
    }
}

using UnityEngine;

#if ENABLE_INPUT_SYSTEM && !ENABLE_LEGACY_INPUT_MANAGER
using UnityEngine.InputSystem;
#endif

public sealed class PickupHolder : MonoBehaviour
{
    private PickupItem _heldItem;
    private int _lastPickupFrame = -1;
    private int _lastDropFrame = -1;

    private void Update()
    {
        if (_heldItem == null || Time.frameCount == _lastPickupFrame)
        {
            return;
        }

        if (WasInteractPressedThisFrame())
        {
            DropHeldItem();
        }
    }

    internal void TryHold(PickupItem item)
    {
        if (item == null ||
            _heldItem != null ||
            Time.frameCount == _lastDropFrame)
        {
            return;
        }

        _heldItem = item;
        _lastPickupFrame = Time.frameCount;
        _heldItem.AttachTo(this);
    }

    internal void DropHeldItem()
    {
        if (_heldItem == null)
        {
            return;
        }

        PickupItem itemToDrop = _heldItem;
        _heldItem = null;
        _lastDropFrame = Time.frameCount;

        itemToDrop.DetachFromHolder();
    }

    private void OnDisable()
    {
        DropHeldItem();
    }

    private static bool WasInteractPressedThisFrame()
    {
#if ENABLE_INPUT_SYSTEM && !ENABLE_LEGACY_INPUT_MANAGER
        Keyboard keyboard = Keyboard.current;

        return keyboard != null && keyboard.eKey.wasPressedThisFrame;
#else
        return Input.GetKeyDown(KeyCode.E);
#endif
    }
}

using UnityEngine;

[RequireComponent(typeof(Rigidbody))]
public sealed class PickupItem : MonoBehaviour, IInteractable
{
    [Header("Interaction Text")]
    [SerializeField] private string _pickupText = "Press E to pick up";
    [SerializeField] private string _dropText = "Press E to drop";

    private Rigidbody _rigidbody;
    private Collider[] _itemColliders;
    private Collider[] _playerColliders;
    private Transform _originalParent;
    private PickupHolder _holder;
    private bool _originalUseGravity;
    private bool _originalIsKinematic;
    private int _lastDroppedFrame = -1;

    public string InteractionText
    {
        get { return _holder == null ? _pickupText : _dropText; }
    }

    private void Awake()
    {
        _rigidbody = GetComponent<Rigidbody>();
        _itemColliders = GetComponentsInChildren<Collider>(true);
    }

    public void Interact()
    {
        if (Time.frameCount == _lastDroppedFrame)
        {
            return;
        }

        if (_holder != null)
        {
            _holder.DropHeldItem();
            return;
        }

        PickupHolder holder = FindPlayerHolder();

        if (holder != null)
        {
            holder.TryHold(this);
        }
    }

    internal void AttachTo(PickupHolder holder)
    {
        _holder = holder;
        _originalParent = transform.parent;
        _originalUseGravity = _rigidbody.useGravity;
        _originalIsKinematic = _rigidbody.isKinematic;

#if UNITY_6000_0_OR_NEWER
        _rigidbody.linearVelocity = Vector3.zero;
#else
        _rigidbody.velocity = Vector3.zero;
#endif
        _rigidbody.angularVelocity = Vector3.zero;
        _rigidbody.useGravity = false;
        _rigidbody.isKinematic = true;

        _playerColliders = holder.transform.root
            .GetComponentsInChildren<Collider>(true);

        SetPlayerCollisionIgnored(true);

        transform.SetParent(holder.transform, false);
        transform.localPosition = Vector3.zero;
        transform.localRotation = Quaternion.identity;
    }

    internal void DetachFromHolder()
    {
        transform.SetParent(_originalParent, true);

        SetPlayerCollisionIgnored(false);

        _rigidbody.isKinematic = _originalIsKinematic;
        _rigidbody.useGravity = _originalUseGravity;

        _holder = null;
        _lastDroppedFrame = Time.frameCount;
    }

    private static PickupHolder FindPlayerHolder()
    {
#if UNITY_2023_1_OR_NEWER
        return FindFirstObjectByType<PickupHolder>();
#else
        return FindObjectOfType<PickupHolder>();
#endif
    }

    private void SetPlayerCollisionIgnored(bool ignored)
    {
        if (_itemColliders == null || _playerColliders == null)
        {
            return;
        }

        for (int itemIndex = 0; itemIndex < _itemColliders.Length; itemIndex++)
        {
            Collider itemCollider = _itemColliders[itemIndex];

            if (itemCollider == null)
            {
                continue;
            }

            for (int playerIndex = 0;
                 playerIndex < _playerColliders.Length;
                 playerIndex++)
            {
                Collider playerCollider = _playerColliders[playerIndex];

                if (playerCollider != null && playerCollider != itemCollider)
                {
                    Physics.IgnoreCollision(
                        itemCollider,
                        playerCollider,
                        ignored);
                }
            }
        }
    }
}

using UnityEditor;

[CustomEditor(typeof(PlayerInteraction))]
public class PlayerInteractEditor : Editor
{
    #region Serialized Properties

    SerializedProperty layerMask;
    SerializedProperty range;
    SerializedProperty sphereCastRadius;
    SerializedProperty interactionText;
    SerializedProperty player;

Engineering

หลักการทำงานเชิงลึก (Deep Dive)

อธิบายแนวคิดเบื้องหลังการออกแบบคณิตศาสตร์และฟิสิกส์ ว่าทำไมระบบจึงทำงานได้ราบรื่นและไม่เกิดข้อขัดแย้ง

1. ฟิสิกส์ Velocity-Change ควบคุมการเคลื่อนที่

ระบบไม่ใช้การบวกค่า Position ดิบ แต่คำนวณส่วนต่างความเร็ว delta = targetVelocity - rb.linearVelocity แล้วส่งผ่าน ForceMode.VelocityChange ทำให้ผู้เล่นเปลี่ยนทิศทางได้ทันทีโดยไม่ไถล และยังรองรับแรงชนจากฟิสิกส์ภายนอกอย่างสมบูรณ์

Vector3 delta = (target - current) * control;
rb.AddForce(delta, ForceMode.VelocityChange);

2. State Suspension สลับสถานะไม่ให้ฟิสิกส์ตีกัน

เมื่อเข้าสู่สถานะสไลด์หรือไต่กำแพง ExtraMovement จะสั่งปิด SimpleMovement.enabled = false ชั่วคราว เพื่อไม่ให้อินพุตเดินปกติมาแย่งแรงส่ง เมื่อหลุดจากกำแพงหรือลุกขึ้นยืนจึงเปิดระบบเดินกลับมา

bool shouldSuspend = isSliding || isWallRunning;
movementScriptToSuspend.enabled = !shouldSuspend;

3. Dual-Lerp Procedural Recoil

การดีดของปืนคำนวณด้วย Lerp สองชั้น ชั้นแรก snappiness ดึงปืนเข้าหาจุดดีดทันที และชั้นสอง recoverySpeed ค่อยๆ คืนตำแหน่งจุดดีดกลับสู่จุดกึ่งกลาง (Vector3.zero) ทำให้ได้ความรู้สึกกระชากแต่คืนตัวนุ่มนวล

// ชั้น 1: วิ่งเข้าหาจุดดีด
current = Vector3.Lerp(current, target, snappiness * dt);
// ชั้น 2: จุดดีดกลับสู่ตำแหน่งปกติ
target = Vector3.Lerp(target, Vector3.zero, recoverySpeed * dt);

4. Physics Handoff & Ignore Collision (Pickup)

ตอนหยิบไอเทม ระบบจะจำค่าเดิมของ useGravity และ isKinematic ไว้ แล้วสั่ง isKinematic = true พร้อมสั่ง Physics.IgnoreCollision กับตัวละคร เพื่อป้องกันแรงชนประหลาดที่อาจดีดผู้เล่นลอย

Physics.IgnoreCollision(itemCollider, playerCollider, true);
transform.SetParent(holder.transform, false);
Guide

วิธีติดตั้งและเริ่มต้นใช้งาน (Setup Guide)

ขั้นตอนการติดตั้งโมดูลและการตั้งค่า Layers ใน Unity Inspector

  1. สร้าง Layers ใน Unity

    ไปที่ Edit > Project Settings > Tags and Layers แล้วเพิ่มเลเยอร์ต่อไปนี้:

    • Ground – สำหรับพื้นผิวที่ผู้เล่นสามารถเดินหรือกระโดดได้
    • Enemy – สำหรับเป้าหมายหรือศัตรูที่รับการยิง Raycast
    • Wall – สำหรับกำแพงที่ต้องการให้วิ่งไต่ได้ (สำหรับโมดูล Movement)
    • Interactable (ตัวเลือกเสริม) – สำหรับวัตถุที่สามารถหยิบหรือตอบสนองได้
  2. ลาก Prefab ลงใน Scene

    แบบพื้นฐาน (Base): ลาก BaseFPSCharacter/Assets/Prefabs/Player.prefab ลงใน Hierarchy แล้วตั้งค่า Ground Mask = Ground และ Hit Layers = Enemy, Default

    แบบตัวละครพร้อมระบบสมบูรณ์ (Variant): ลาก Visual System/Assets/Prefabs/Player Variant.prefab แล้วตรวจสอบค่าใน Inspector:

    • SimpleMovement: Ground Mask = Ground
    • ExtraMovement: Ground Mask = Ground, Wall Mask = Wall, ลากคอมโพเนนต์ SimpleMovement มาใส่ในช่อง Movement Script To Suspend
    • GunFeel: Hit Layers = Ground, Enemy, Default
  3. ตั้งค่าระบบ Pick up & Interaction

    สร้าง Empty GameObject ใต้กล้องของผู้เล่น ตั้งชื่อว่า HoldPoint แล้วแนบสคริปต์ PickupHolder.cs ส่วนวัตถุในฉากที่ต้องการให้หยิบได้ ให้แนบคอมโพเนนต์ PickupItem.cs และ Rigidbody
